! SW-CORE (Switch-PT com 3 modulos PT-SWITCH-NM-1FGE nos slots 1, 2 e 3)
! Distribui as VLANs do roteador para os 3 switches de acesso por fibra.
! Senhas de laboratorio, validas so no Packet Tracer.
enable
configure terminal
hostname SW-CORE
no ip domain-lookup
enable secret CLenable2026
service password-encryption
banner motd # Acesso restrito - Posto Caminho Livre #
!
vlan 10
 name EQUIPE
vlan 20
 name CIDADAO_WIFI
vlan 30
 name SERVIDORES
vlan 99
 name GERENCIA
vlan 999
 name NATIVA_SEM_USO
exit
!
interface FastEthernet0/1
 description UTP TRUNK para R-POSTO
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,20,30,99,999
interface GigabitEthernet1/1
 description FIBRA TRUNK para SW-EQUIPE
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 10,99,999
interface GigabitEthernet2/1
 description FIBRA TRUNK para SW-SERVIDORES
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99,999
interface GigabitEthernet3/1
 description FIBRA TRUNK para SW-CIDADAO
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 20,99,999
interface FastEthernet4/1
 description SEM USO
 switchport mode access
 switchport access vlan 999
 shutdown
interface FastEthernet5/1
 description SEM USO
 switchport mode access
 switchport access vlan 999
 shutdown
!
interface Vlan1
 shutdown
interface Vlan99
 description GERENCIA
 ip address 192.168.10.194 255.255.255.240
 no shutdown
exit
ip default-gateway 192.168.10.193
!
line console 0
 password CLconsole2026
 login
 exec-timeout 5 0
end
write memory
