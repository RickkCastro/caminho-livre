! SW-SERVIDORES (Switch-PT com 1 modulo PT-SWITCH-NM-1FGE no slot 1)
! Servidores do posto na VLAN 30: SRV-POSTO (DNS + web) e SRV-EMAIL.
! Senhas de laboratorio, validas so no Packet Tracer.
enable
configure terminal
hostname SW-SERVIDORES
no ip domain-lookup
enable secret CLenable2026
service password-encryption
banner motd # Acesso restrito - Posto Caminho Livre #
!
vlan 30
 name SERVIDORES
vlan 99
 name GERENCIA
vlan 999
 name NATIVA_SEM_USO
exit
!

 description FIBRA TRUNK para SW-CORE
 switchport mode trunk
 switchport trunk native vlan 999
 switchport trunk allowed vlan 30,99,999
interface FastEthernet0/1
 description SRV-POSTO
 switchport mode access
 switchport access vlan 30
interface FastEthernet2/1
 description SRV-EMAIL
 switchport mode access
 switchport access vlan 30
interface FastEthernet3/1
 description SEM USO
 switchport mode access
 switchport access vlan 999
 shutdown
interface FastEthernet4/1
 description SEM USO
 switchport mode access
 switchport access vlan 999
 shutdown
interface FastEthernet5/1
 description SEM USO
 switchport mode access
 switchport access vlan 999
 shutdown
!
interface Vlan1
 shutdown

 description GERENCIA
 ip address 192.168.10.196 255.255.255.240
 no shutdown
exit
